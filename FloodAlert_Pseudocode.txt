Function Main
    Declare String Array residentName[500]
    Declare String Array residentPhone[500]
    Declare String Array residentCommunity[500]
    Declare Integer residentCount
    Declare Integer Array stats[2]

    Assign residentCount = 0
    Assign stats[0] = 0
    Assign stats[1] = 0
    Assign residentCount = loadSampleResidents(residentName, residentPhone, residentCommunity)
    Declare String community
    Declare String risk
    Declare Real rainfall
    Declare Real riverLevel
    Declare Integer index
    Declare Integer sent

    Output "Enter community name: "
    Input community
    Output "Enter rainfall in the last hour (mm): "
    Input rainfall
    While rainfall < 0
        Output "Invalid - rainfall cannot be negative. Re-enter: "
        Input rainfall
    End
    Output "Enter drain/river water level (m): "
    Input riverLevel
    While riverLevel < 0
        Output "Invalid - water level cannot be negative. Re-enter: "
        Input riverLevel
    End
    If rainfall >= 50 || riverLevel >= 2.5
        Assign risk = "RED"
    Else
        If rainfall >= 30 && riverLevel >= 1.5
            Assign risk = "ORANGE"
        Else
            If rainfall >= 30 || riverLevel >= 1.5
                Assign risk = "YELLOW"
            Else
                Assign risk = "GREEN"
            End
        End
    End
    Output "Community: " & community
    Output "Risk level: " & risk
    Output getMessage(risk)
    If risk == "GREEN"
        Output "No alert needed."
    Else
        Assign sent = 0
        For index = 0 to residentCount - 1
            If residentCommunity[index] == community
                Output "SMS to " & residentPhone[index]
                Assign sent = sent + 1
            End
        End
        Output sent & " residents alerted in " & community
    End
End

Function getMessage (String risk)
    Declare String message

    If risk == "RED"
        Assign message = "RED ALERT: Flooding expected. Move to higher ground NOW and follow NDMA instructions."
    Else
        If risk == "ORANGE"
            Assign message = "ORANGE WARNING: High flood risk. Prepare to move. Keep children away from drains."
        Else
            If risk == "YELLOW"
                Assign message = "YELLOW ALERT: Heavy rain or rising water. Stay alert and clear your drains."
            Else
                Assign message = "GREEN: Normal conditions."
            End
        End
    End
    Return String message
End
